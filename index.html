Feature: E-Commerce Checkout Flow

  Scenario: Complete purchase with valid payment
    Given user opens shopping cart with items
    When user clicks proceed to checkout
    And fills payment credentials
    Then order confirmation modal is displayed
import { test, expect } from '@playwright/test';
import { CheckoutPage } from './pages/CheckoutPage';

test('Complete purchase flow', async ({ page }) => {
  const checkout = new CheckoutPage(page);
  await checkout.navigate();
  await checkout.proceedToPayment();
  await expect(checkout.confirmationModal).toBeVisible();
});
import { Page, Locator } from '@playwright/test';

export class CheckoutPage {
  readonly page: Page;
  readonly checkoutBtn: Locator;
  readonly confirmationModal: Locator;

  constructor(page: Page) {
    this.page = page;
    this.checkoutBtn = page.getByRole('button', { name: 'Checkout' });
    this.confirmationModal = page.getByTestId('order-success');
  }
}
